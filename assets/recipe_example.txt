nwndgen-recipe 1

# Example recipe using every key. Generate the donor files next to it first (needs a NWN install):
#   build/nwnpack tilepack-kit assets/tilepack_tdc01_kit.txt
#   build/nwnpack templates assets
# then: build/nwndgen --game nwn1 --base BASE.mod --recipe assets/recipe_example.txt --output out.mod

# Required: what to build from. Paths are relative to this file.
# Tile pool; the kit pack is needed for "corridors kit".
tilepack tilepack_tdc01_kit.txt
# Donor area: header, lighting, weather.
template_are tpl_are.are
# Donor GIT: area properties plus a stock transition trigger.
template_git tpl_git.git
module_name The Shifting Cellars
module_tag SHIFTING_CELLARS
# min level, max level, min party, max party this recipe accepts.
support 1 6 1 4

# Required: the levels, chained in order (a -> b -> c).
# area <resref> <width> <height> <rooms> <display name>
area scel_a 14 14 3 Shifting Cellars - Upper
area scel_b 14 14 3 Shifting Cellars - Middle
area scel_c 16 16 4 Shifting Cellars - Lower

# Optional: module text.
module_description Generated by nwndgen. Rooms joined by stock corridors with doors.

# Optional: corridor style (wide is the default, kit uses the stock pieces).
corridors kit
# Optional: none (default), plain (door type 22), mixed (plain plus the second corridor door look, type 23)
# or doorway (a door of one of seven generic looks in every corridor entrance).
# Needs corridors kit and template_door.
doors mixed
template_door tpl_door.git

# Optional: creatures, boss, vault, loot and scenery. roster and scripts go together.
roster roster.txt
scripts scripts
# full (default), boss_only or loot_test.
spawn_mode full
# 0-5 stock AI henchmen join the first player.
henchmen 2
# Optional: how much creatures carry, low, mid (default) or high.
loot_amount mid

# Optional: furnish rooms by type from the roster's roomprop lines, off (default), low, mid or high.
dressing off

# Optional: props along the stock kit corridors (needs corridors kit and a pack from the current nwnpack tilepack-kit), off (default), low, mid or high.
corridor_dressing off

# Optional: lock some corridor doors (pickable, or keyed with the key on a creature in an earlier room) and some chests (needs corridors kit and doors); off (default), low, mid or high.
locks off

# Optional: hide some room creatures until a chest in their room is opened or a player comes near; off (default), low, mid or high.
ambush off

# Optional: hidden floor traps and trapped chests, strength by character level; off (default), low, mid or high.
traps off

# Optional: turn one door per area into a gate that a lever in an earlier room opens; off (default) or on. Needs locks and the roster's lever line.
levers off

# Optional: a few room creatures walk a short route inside their room; off (default), low, mid or high.
patrols off

# Optional: random encounters at play time (when resting, and after a long quiet spell); off (default), low, mid or high.
encounters off

# Optional: the dungeon answers the player (death alerts, vault reinforcements, the boss calls for help); off (default) or on.
reactions off
# Test aid: on gives every creature the scripts create (not henchmen) 1 hit point. Needs roster and scripts.
one_hp off

# Optional run defaults. The command-line options of the same name override them.
seed 12345
party_size 1
min_level 4
max_level 5
