# Example recipe using every key. Generate the donor files next to it first (needs a NWN install):
#   build/nwnpack tilepack-kit assets/tilepack_tdc01_kit.txt
#   build/nwnpack templates assets
# then: build/nwndgen --game nwn1 --base BASE.mod --recipe assets/recipe_example.txt --output out.mod

# Required: what to build from. Paths are relative to this file.
# Tile pool; the kit pack is needed for "corridors kit".
tilepack tilepack_tdc01_kit.txt
# Donor area: header, lighting, weather.
template_are tpl_are.are
# Donor GIT: area properties plus a stock transition trigger.
template_git tpl_git.git
module_name The Shifting Cellars
module_tag SHIFTING_CELLARS
# min level, max level, min party, max party this recipe accepts.
support 1 6 1 4

# Required: the levels, chained in order (a -> b -> c).
# area <resref> <width> <height> <rooms> <display name>
area scel_a 14 14 3 Shifting Cellars - Upper
area scel_b 14 14 3 Shifting Cellars - Middle
area scel_c 16 16 4 Shifting Cellars - Lower

# Optional: module text.
module_description Generated by nwndgen. Rooms joined by stock corridors with doors.

# Optional: corridor style (wide is the default, kit uses the stock pieces).
corridors kit
# Optional: none (default), plain (door type 22) or mixed (plain plus the second corridor door look, type 23).
# Needs corridors kit and template_door.
doors mixed
template_door tpl_door.git

# Optional: creatures, boss, vault, loot and scenery. roster and scripts go together.
roster roster.txt
scripts scripts
# full (default), boss_only or loot_test.
spawn_mode full
# 0-5 stock AI henchmen join the first player.
henchmen 2
# Optional: how much creatures carry, low, mid (default) or high.
loot_amount mid

# Optional run defaults. The command-line options of the same name override them.
seed 12345
party_size 1
min_level 4
max_level 5
